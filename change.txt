-- ============================================
-- BAN USER FUNCTION (Admin Only)
-- ============================================
CREATE OR REPLACE FUNCTION ban_user(
  target_user_id UUID,
  reason TEXT DEFAULT NULL
)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  current_user_id UUID := auth.uid();
BEGIN
  -- Check if caller is admin
  IF NOT EXISTS (
    SELECT 1 FROM public.profiles 
    WHERE id = current_user_id AND role = 'admin'
  ) THEN
    RAISE EXCEPTION 'Only admins can ban users';
  END IF;
  
  -- Prevent admin from banning themselves
  IF target_user_id = current_user_id THEN
    RAISE EXCEPTION 'You cannot ban yourself';
  END IF;
  
  -- Ban the user
  UPDATE public.profiles
  SET 
    is_banned = true,
    ban_reason = reason,
    banned_at = NOW(),
    updated_at = NOW()
  WHERE id = target_user_id;
  
  -- Revoke all user sessions
  DELETE FROM auth.sessions WHERE user_id = target_user_id;
  
  -- Log the ban
  INSERT INTO public.admin_logs (
    action,
    target_user_id,
    details,
    performed_by
  ) VALUES (
    'user_ban',
    target_user_id,
    jsonb_build_object('reason', reason),
    current_user_id
  );
END;
$$;

-- ============================================
-- REQUEST ACCOUNT DELETION (Soft Delete - No Hard Delete)
-- User requests, admin approve
-- ============================================
CREATE OR REPLACE FUNCTION request_account_deletion()
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  current_user_id UUID := auth.uid();
BEGIN
  -- Check if user is logged in
  IF current_user_id IS NULL THEN
    RAISE EXCEPTION 'You must be logged in to request account deletion';
  END IF;
  
  -- Mark account for deletion (soft delete)
  UPDATE public.profiles
  SET 
    deletion_requested_at = NOW(),
    updated_at = NOW()
  WHERE id = current_user_id;
  
  -- Log the deletion request
  INSERT INTO public.admin_logs (
    action,
    target_user_id,
    details,
    performed_by
  ) VALUES (
    'deletion_requested',
    current_user_id,
    jsonb_build_object(),
    current_user_id
  );
END;
$$;

-- ============================================
-- APPROVE ACCOUNT DELETION (Admin Only - Soft Delete Only)
-- ============================================
CREATE OR REPLACE FUNCTION approve_account_deletion(
  target_user_id UUID
)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  current_user_id UUID := auth.uid();
BEGIN
  -- Check if caller is admin
  IF NOT EXISTS (
    SELECT 1 FROM public.profiles 
    WHERE id = current_user_id AND role = 'admin'
  ) THEN
    RAISE EXCEPTION 'Only admins can approve account deletions';
  END IF;
  
  -- Soft delete the account (mark as deleted, don't remove from database)
  UPDATE public.profiles
  SET 
    deleted_at = NOW(),
    deletion_requested_at = NULL,
    updated_at = NOW()
  WHERE id = target_user_id;
  
  -- Log the deletion approval
  INSERT INTO public.admin_logs (
    action,
    target_user_id,
    details,
    performed_by
  ) VALUES (
    'deletion_approved',
    target_user_id,
    jsonb_build_object(),
    current_user_id
  );
END;
$$;

-- ============================================
-- CANCEL DELETION REQUEST (User Can Cancel Before Admin Approves)
-- ============================================
CREATE OR REPLACE FUNCTION cancel_deletion_request()
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  current_user_id UUID := auth.uid();
BEGIN
  -- Check if user is logged in
  IF current_user_id IS NULL THEN
    RAISE EXCEPTION 'You must be logged in to cancel deletion request';
  END IF;
  
  -- Cancel the deletion request
  UPDATE public.profiles
  SET 
    deletion_requested_at = NULL,
    updated_at = NOW()
  WHERE id = current_user_id AND deleted_at IS NULL;
END;
$$;

    WHERE id = current_user_id AND role = 'admin'
  ) THEN
    RAISE EXCEPTION 'Only admins can ban users';
  END IF;
  
  -- Prevent admin from banning themselves
  IF target_user_id = current_user_id THEN
    RAISE EXCEPTION 'You cannot ban yourself';
  END IF;
  
  -- Ban the user
  UPDATE public.profiles
  SET 
    is_banned = true,
    ban_reason = reason,
    banned_at = NOW(),
    updated_at = NOW()
  WHERE id = target_user_id;
  
  -- Revoke all user sessions
  DELETE FROM auth.sessions WHERE user_id = target_user_id;
  
  -- Log the ban
  INSERT INTO public.admin_logs (
    action,
    target_user_id,
    details,
    performed_by
  ) VALUES (
    'user_ban',
    target_user_id,
    jsonb_build_object('reason', reason),
    current_user_id
  );
END;
$$;

-- UNBAN USER FUNCTION (Admin Only)
-- ============================================
CREATE OR REPLACE FUNCTION unban_user(
  target_user_id UUID
)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE

  ) VALUES (
    'user_unban',
    target_user_id,
    jsonb_build_object(),

  );

$$;

-- ============================================
-- DELETE ACCOUNT FUNCTION (User or Admin)
-- ============================================
CREATE OR REPLACE FUNCTION delete_account(
  target_user_id UUID DEFAULT NULL
)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  current_user_id UUID := auth.uid();
  user_to_delete UUID := COALESCE(target_user_id, current_user_id);
BEGIN
  -- If no target_user_id provided, user is deleting their own account
  IF target_user_id IS NULL THEN
    user_to_delete := current_user_id;
  END IF;
  
  -- Check permissions
  IF target_user_id IS NOT NULL THEN
    -- Admin deleting another user
    IF NOT EXISTS (
      SELECT 1 FROM public.profiles 
      WHERE id = current_user_id AND role = 'admin'
    ) THEN
      RAISE EXCEPTION 'Only admins can delete other users';
    END IF;
  ELSE
    -- User deleting own account - check authentication
    IF current_user_id IS NULL THEN
      RAISE EXCEPTION 'You must be logged in to delete your account';
    END IF;
  END IF;
  
  -- Prevent admin from deleting themselves
  IF user_to_delete = current_user_id AND target_user_id IS NULL THEN
    -- This is fine (user deleting own account)
  ELSIF user_to_delete = current_user_id THEN
    RAISE EXCEPTION 'You cannot delete your own account. Contact support.';
  END IF;
  
  -- Mark as deleting (soft delete first)
  UPDATE public.profiles
  SET 
    deleted_at = NOW(),
    updated_at = NOW()
  WHERE id = user_to_delete;
  
  -- Delete user data from related tables (cascade will handle most)
  DELETE FROM public.notifications WHERE user_id = user_to_delete;
  DELETE FROM public.subscriptions WHERE user_id = user_to_delete;
  DELETE FROM public.admin_logs WHERE target_user_id = user_to_delete;
  
  -- Delete from auth.users (hard delete)
  DELETE FROM auth.users WHERE id = user_to_delete;
  
  -- Log the deletion
  INSERT INTO public.admin_logs (
    action,
    target_user_id,
    details,
    performed_by
  ) VALUES (
    'account_delete',
    user_to_delete,
    jsonb_build_object(
      'deleted_by', 
      CASE WHEN target_user_id IS NULL THEN 'self' ELSE 'admin' END
    ),
    current_user_id
  );
END;
$$;
